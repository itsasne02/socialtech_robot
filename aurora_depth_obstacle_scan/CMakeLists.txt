cmake_minimum_required(VERSION 3.16)
project(aurora_depth_obstacle_scan)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
add_library(depth_projection src/projection.cpp)
target_compile_features(depth_projection PUBLIC cxx_std_17)
target_include_directories(depth_projection PUBLIC include)
target_compile_options(depth_projection PRIVATE -Wall -Wextra -Wpedantic)
add_executable(aurora_depth_obstacle_scan src/node.cpp)
target_compile_options(aurora_depth_obstacle_scan PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(aurora_depth_obstacle_scan depth_projection)
ament_target_dependencies(aurora_depth_obstacle_scan rclcpp rcl_interfaces sensor_msgs tf2 tf2_ros)
install(TARGETS aurora_depth_obstacle_scan DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch rviz DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_projection test/test_projection.cpp)
  target_link_libraries(test_projection depth_projection)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME test_ros_contract COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/test/test_ros_contract.py $<TARGET_FILE:aurora_depth_obstacle_scan>)
  set_tests_properties(test_ros_contract PROPERTIES TIMEOUT 40
    ENVIRONMENT "ROS_DOMAIN_ID=199;ROS_AUTOMATIC_DISCOVERY_RANGE=LOCALHOST")
endif()
ament_package()
