cmake_minimum_required(VERSION 3.16)
project(aurora_depth_validation LANGUAGES CXX)
set(AURORA_SDK_ROOT "" CACHE PATH "Existing aurora_remote_public directory")
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
  set(SDK_PLATFORM linux_aarch64)
elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64)$")
  set(SDK_PLATFORM linux_x86_64)
else()
  message(FATAL_ERROR "Unsupported SDK platform: ${CMAKE_SYSTEM_PROCESSOR}")
endif()
find_path(SDK_INCLUDE aurora_pubsdk_inc.h PATHS "${AURORA_SDK_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
find_library(SDK_LIBRARY slamtec_aurora_remote_sdk PATHS "${AURORA_SDK_ROOT}/lib/${SDK_PLATFORM}" NO_DEFAULT_PATH REQUIRED)
find_package(Threads REQUIRED)
add_executable(sdk_probe sdk_probe.cpp)
target_compile_features(sdk_probe PRIVATE cxx_std_17)
target_compile_options(sdk_probe PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(sdk_probe SYSTEM PRIVATE "${SDK_INCLUDE}")
target_link_libraries(sdk_probe PRIVATE "${SDK_LIBRARY}" Threads::Threads)
